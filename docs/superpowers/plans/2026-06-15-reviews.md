# Reviews Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let a customer leave a 1–5 star review (+ optional comment) on the booking page once the booking is completed, and surface the average rating + recent reviews on the public tenant page.

**Architecture:** Reuse the existing `Review` entity (one row per booking, created on completion by `ReviewsService.requestForBooking`). Add a booking-id-based submit path (auth = possession of the booking id) instead of the token page. Expose review state in the public booking response and an aggregate (+ recent reviews) in the public tenant payload. Frontend renders a review block on `/b/[id]` and a ratings/reviews display on the tenant page.

**Tech Stack:** NestJS + TypeORM + Jest (backend, `nestjs-project`); Next.js App Router + Tailwind v4 (frontend, `bookup.uz`). Backend node via `~/.nvm/versions/node/v24.14.0/bin` (`export PATH="$HOME/.nvm/versions/node/v24.14.0/bin:$PATH"`).

```typescript
  /** Submit a review for a completed booking (auth = possession of the booking id). */
  async submitForBooking(
    businessId: string,
    bookingId: string,
    dto: SubmitReviewDto,
  ): Promise<ReviewState> {
    const rating = dto.rating;
    if (!Number.isInteger(rating) || rating < 1 || rating > 5) {
      throw new AppException(HttpStatus.BAD_REQUEST, ErrorCode.INVALID_REVIEW, 'rating must be an integer 1–5');
    }
    const comment = dto.comment?.trim() || null;
    if (comment && comment.length > REVIEW_COMMENT_MAX) {
      throw new AppException(HttpStatus.BAD_REQUEST, ErrorCode.INVALID_REVIEW, `comment must be at most ${REVIEW_COMMENT_MAX} characters`);
    }

    const booking = await this.bookings.findOne({ where: { id: bookingId, businessId } });
    if (!booking) {
      throw new AppException(HttpStatus.NOT_FOUND, ErrorCode.BOOKING_NOT_FOUND, 'Booking not found');
    }
    if (booking.status !== BookingStatus.COMPLETED) {
      throw new AppException(HttpStatus.BAD_REQUEST, ErrorCode.INVALID_REVIEW, 'Only a completed booking can be reviewed');
    }

    await this.reviews.manager.transaction(async (em) => {
      let review = await em.findOne(Review, {
        where: { bookingId },
        lock: { mode: 'pessimistic_write' },
      });
      if (review && review.submittedAt !== null) {
        throw new AppException(HttpStatus.CONFLICT, ErrorCode.REVIEW_ALREADY_SUBMITTED, 'This review has already been submitted');
      }
      if (!review) {
        review = em.create(Review, {
          bookingId,
          businessId: booking.businessId,
          branchId: booking.branchId,
          userId: booking.userId,
          token: randomUUID().replace(/-/g, ''),
        });
      }
      review.rating = rating;
      review.comment = comment;
      review.submittedAt = new Date();
      await em.save(review);
    });

    return { submitted: true, rating, comment };
  }

  /** Review state for a booking (for the public booking page). Null if no row yet. */
  async stateForBooking(bookingId: string): Promise<ReviewState | null> {
    const review = await this.reviews.findOne({ where: { bookingId } });
    if (!review) return null;
    return { submitted: review.submittedAt !== null, rating: review.rating, comment: review.comment };
  }

  /** Public aggregate + recent submitted reviews for a business (no owner auth). */
  async publicSummary(
    businessId: string,
    limit = 20,
  ): Promise<{ averageRating: number | null; reviewCount: number; reviews: BusinessReview[] }> {
    const rows = await this.reviews
      .createQueryBuilder('rv')
      .leftJoinAndSelect('rv.user', 'u')
      .leftJoinAndSelect('rv.booking', 'b')
      .leftJoinAndSelect('b.items', 'it')
      .leftJoinAndSelect('it.offering', 'o')
      .leftJoinAndSelect('o.service', 's')
      .where('rv.business_id = :businessId', { businessId })
      .andWhere('rv.submitted_at IS NOT NULL')
      .orderBy('rv.submitted_at', 'DESC')
      .addOrderBy('it.position', 'ASC')
      .getMany();

    const reviews: BusinessReview[] = rows.map((rv) => ({
      id: rv.id,
      branchId: rv.branchId,
      rating: rv.rating,
      comment: rv.comment,
      submittedAt: rv.submittedAt ? rv.submittedAt.toISOString() : null,
      // Public: first name only (privacy); guest → guest first name or "Mehmon".
      customerName: rv.user
        ? rv.user.firstName
        : (rv.booking?.guestName?.split(/\s+/)[0] ?? 'Mehmon'),
      services: (rv.booking?.items ?? [])
        .map((it) => it.offering?.service?.name)
        .filter((n): n is LocalizedText => !!n),
    }));
    const rated = reviews.filter((r) => r.rating !== null);
    const averageRating =
      rated.length === 0 ? null
      : Math.round((rated.reduce((s, r) => s + (r.rating ?? 0), 0) / rated.length) * 10) / 10;
    return { averageRating, reviewCount: reviews.length, reviews: reviews.slice(0, limit) };
  }
```

```typescript
export async function submitReviewAction(
  subdomain: string,
  bookingId: string,
  rating: number,
  comment: string,
): Promise<{ ok: true } | { ok: false; error: string }> {
  try {
    const res = await serverFetch(
      `${API_BASE}/public/tenants/${encodeURIComponent(subdomain)}/bookings/${encodeURIComponent(bookingId)}/review`,
      {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ rating, comment: comment.trim() || undefined }),
      },
    );
    if (!res.ok) return { ok: false, error: await errorMessage(res) };
    return { ok: true };
  } catch {
    return { ok: false, error: NETWORK_ERROR_UZ };
  }
}
```

```tsx
'use client';

import { useState } from 'react';
import { useRouter } from 'next/navigation';
import { Star } from 'lucide-react';
import type { ResultDict } from '@/lib/dictionaries/result';
import { submitReviewAction } from './actions';

export function ReviewBlock({
  subdomain,
  bookingId,
  dict,
  initial,
}: {
  subdomain: string;
  bookingId: string;
  dict: ResultDict;
  /** Existing review state for this booking; null when no row yet. */
  initial: { submitted: boolean; rating: number | null; comment: string | null } | null;
}) {
  const router = useRouter();
  const submitted = !!initial?.submitted;
  const [rating, setRating] = useState(initial?.rating ?? 0);
  const [hover, setHover] = useState(0);
  const [comment, setComment] = useState(initial?.comment ?? '');
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [done, setDone] = useState(submitted);

  const Stars = ({ readOnly }: { readOnly?: boolean }) => (
    <div className="flex gap-1.5">
      {[1, 2, 3, 4, 5].map((n) => {
        const filled = (hover || rating) >= n;
        return (
          <button
            key={n}
            type="button"
            disabled={readOnly || busy}
            aria-label={`${n}`}
            onMouseEnter={() => !readOnly && setHover(n)}
            onMouseLeave={() => !readOnly && setHover(0)}
            onClick={() => !readOnly && setRating(n)}
            className="transition-transform active:scale-90 disabled:cursor-default"
          >
            <Star size={32} className={filled ? 'fill-amber-400 text-amber-400' : 'text-border'} />
          </button>
        );
      })}
    </div>
  );

  const submit = async () => {
    if (rating < 1 || busy) return;
    setBusy(true);
    setError(null);
    const r = await submitReviewAction(subdomain, bookingId, rating, comment);
    if (r.ok) { setDone(true); router.refresh(); return; }
    setBusy(false);
    setError(r.error);
  };

  if (done) {
    return (
      <div className="mt-6 rounded-2xl border border-border bg-card p-5">
        <p className="text-[15px] font-bold text-foreground">{dict.reviewThanks}</p>
        <div className="mt-3"><Stars readOnly /></div>
        {(initial?.comment || comment) && (
          <p className="mt-3 text-sm text-muted-foreground">{initial?.comment || comment}</p>
        )}
      </div>
    );
  }

  return (
    <div className="mt-6 rounded-2xl border border-border bg-card p-5">
      <p className="text-[15px] font-bold text-foreground">{dict.reviewTitle}</p>
      <p className="mt-3 text-sm text-muted-foreground">{dict.reviewRatingLabel}</p>
      <div className="mt-2"><Stars /></div>
      <label className="mt-4 block text-sm text-muted-foreground">{dict.reviewCommentLabel}</label>
      <textarea
        value={comment}
        onChange={(e) => setComment(e.target.value)}
        rows={3}
        maxLength={2000}
        className="mt-2 w-full resize-none rounded-xl border border-border bg-card p-3 text-sm text-foreground outline-none transition-colors focus:border-foreground"
      />
      {error && <p className="mt-2 text-sm font-semibold text-red-600">{error}</p>}
      <button
        type="button"
        onClick={submit}
        disabled={rating < 1 || busy}
        className="mt-4 h-12 w-full rounded-xl bg-foreground text-sm font-bold text-background transition-opacity hover:opacity-90 disabled:opacity-40"
      >
        {dict.reviewSubmit}
      </button>
    </div>
  );
}
```
